在 Python 中检查边界。该示例使用了图表中的 12 个开盘时间，并在 Asia/Hong_Kong 时区设置了时间戳和批次边界。每个断言都按（第一批次，下一批次）的顺序检查计数值。

```python
from datetime import datetime
from zoneinfo import ZoneInfo
hong_kong = ZoneInfo('Asia/Hong_Kong')
def at(hhmm): return datetime.fromisoformat(f'2026-11-14T{hhmm}').replace(tzinfo=hong_kong)
opens = [at(hhmm) for hhmm in ( '12:30', '12:35', '12:40', '12:45', '12:50', '12:55', '13:00', '13:05', '13:10', '13:15', '13:20', '13:25', )]
start, boundary, end = at('12:30'), at('13:00'), at('13:30')
def boundary_counts(rows, *, include_first_end=False):
    if include_first_end:
        first = [t for t in rows if start <= t <= boundary]
    else:
        first = [t for t in rows if start <= t < boundary]
    second = [t for t in rows if boundary <= t < end]
    return first.count(boundary), second.count(boundary)
assert boundary_counts(opens) == (0, 1)
assert boundary_counts(opens, include_first_end=True) == (1, 1)
assert boundary_counts([t for t in opens if t != boundary]) == (0, 0)
assert boundary_counts(opens + [boundary]) == (0, 2)
```
